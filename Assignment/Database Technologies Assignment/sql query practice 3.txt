Department-wise Employee Count

Create a procedure that displays each department along with the number of employees in it.

drop procedure if exist displayDeptEmp;

delimiter //

create procedure displayDeptEmp()
begin
select count(*), deptno from emp group by deptno;
end//

delimiter ;



Salary Increment Procedure

Write a procedure that increases the salary of all employees in a department by a given percentage.

Parameters: deptno, percent_increment.


drop procedure if exists incSal;
delimiter //
create procedure incSal(in pdeptno int , ppercent int)
begin
update emp
set sal = sal + (sal * (ppercent/100))
where deptno = pdeptno;
end//

delimiter;



Conditional Procedure

Write a procedure that takes an employee number as input and displays:

“High Salary” if salary > 50000

“Medium Salary” if salary between 30000–50000

“Low Salary” if salary < 30000


drop procedure if exists dispSalStatus;
delimiter //
create procedure dispSalStatus(in pempno int)
begin 
select empno, ename, sal, 
case 
when sal >=6000 then "Highest salary"
when sal >=2000 and sal<6000 then "Medium salary"
else "Lowest Salary"
end as Status from emp
where empno = pempno;
end//

delimiter ;



Employee Promotion Procedure

Write a procedure to promote employees:

If job = ‘Clerk’, change job = ‘Senior Clerk’

If job = ‘Analyst’, change job = ‘Senior Analyst’

Update table and show affected rows.


drop procedure if exists updateJob;
delimiter //
create procedure updateJob()
begin
update emp
set job = 'S.Clerk' where job = 'clerk';

update emp 
set job = 'S.Analyat' where job = 'analyst';

select empno, ename, job, sal, deptno from emp;

end//

delimiter ;

drop procedure if exists updateDiscount;

delimiter //
create procedure updateDiscount()
begin

declare vbookid int;
declare vpriceperday, vdiscount double(9,2);

declare vset int default 0;

declare updatecur cursor for select booking_id,costPerDay from hotel_booking;
declare continue handler for NOT FOUND set vset = 1;

open updatecur;
label1:loop
fetch updatecur into vbookid,vpriceperday;
if vset = 1 then
leave label1;
end if;

if vpriceperday <= 4500 then
	set vdiscount = 0;
elseif vpriceperday > 4500 and vpriceperday<=10000 then
	set vdiscount = vpriceperday * 0.10;
	
elseif vpriceperday>10000 then	
	set vdiscount = vpriceperday*0.20;
else
	set vdiscount= 0;
	
end if;

update hotel_booking 
set discount = vdiscount where booking_id = vbookid;

end loop;

close updatecur;

end//

delimiter ;

drop function if exists calcPrice;

delimiter //

create function calcPrice(bookid int)
returns double(9,2)
deterministic
begin

declare vcheckin date;
declare vcheckout date;
declare vcostperday double(9,2);
declare vprice double(9,2) default 0;
declare daycount int;

select costperday, check_in_date, check_out_date into vcostperday, vcheckin, vcheckout from hotel_booking where booking_id = bookid;

set daycount = datediff(vcheckout,vcheckin);
set vprice = vcostperday * daycount;
return vprice;

end//

delimiter ;

returns double(9,2)
deterministic
begin

declare si double(9,2) default 0;

set si = (principle*rateofinterest*noofyear)/100;

return si;

end//
