drop procedure if exists updateSalary;

create procedure updateSalary()
begin

declare vempno, vset int default 0;
declare vename, vjob varchar(20);
declare vHIREDATE date;
declare vsal, vnewsal double(9,2);

declare salcursor cursor for select empno, ename, job, sal, HIREDATE from emp;
declare continue handler for NOT FOUND set vset = 1;

open salcursor;

label1:loop
fetch salcursor into vempno, vename, vjob, vsal, vHIREDATE;
if vset = 1 then
leave label1;
end if;

if timestampdiff(year,vHIREDATE,curdate())<=35 then
	set vnewsal = vsal;
	
elseif timestampdiff(year,vHIREDATE,curdate())>35 and timestampdiff(year,vHIREDATE,curdate())<=38 then
	set vnewsal = vsal + (vsal*0.2);
	
elseif timestampdiff(year,vHIREDATE,curdate())>38 then
	set vnewsal = vsal + (vsal*0.25);

else 
	set vnewsal=vsal;
end if;

update emp
set sal = vnewsal where empno = vempno;
end loop;
end//

CREATE PROCEDURE updateSalary()
BEGIN
    DECLARE vempno INT;
    DECLARE vhiredate DATE;
    DECLARE vsal DECIMAL(10,2);
    DECLARE vexp INT;

    DECLARE done INT DEFAULT 0;

    DECLARE salcursor CURSOR FOR
        SELECT empno, hiredate, sal FROM emp;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN salcursor;

    read_loop: LOOP
        FETCH salcursor INTO vempno, vhiredate, vsal;
        IF done THEN
            LEAVE read_loop;
        END IF;

        -- Calculate years of experience
        SET vexp = TIMESTAMPDIFF(YEAR, vhiredate, CURDATE());

        -- Apply salary increment rules
        IF vexp > 35 AND vexp <= 38 THEN
            SET vsal = vsal + (vsal * 0.20);
        ELSEIF vexp > 38 THEN
            SET vsal = vsal + (vsal * 0.25);
        END IF;

        -- Update the salary in emp table
        UPDATE emp SET sal = vsal WHERE empno = vempno;
    END LOOP;

    CLOSE salcursor;
END//

DELIMITER ;
