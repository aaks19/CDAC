drop procedure if exists updateSalary;

delimiter //
create procedure updateSalary()
begin

declare vempno int;
declare vhiredate date;
declare vsal decimal(9,2);
declare vnewsal decimal(9,2);
declare vexp int;
declare vename, vjob varchar(20);

declare vset int default 0;

declare salcursor cursor for select empno,ename, job, sal, hiredate from emp;

declare continue handler for NOT FOUND set vset = 1;

open salcursor ;

label1:loop
fetch salcursor into vempno, vename, vjob, vsal, vhiredate;

if vset = 1 then
leave label1;
end if;



set vexp = timestampdiff(year,vhiredate,curdate());

if vexp>35 and vexp<=38 then
set vnewsal = vsal + (vsal*0.20);

elseif vexp>38 then
set vnewsal = vsal + (vsal*0.25);

end if;

update emp
set sal = vnewsal where empno = vempno;

end loop;

close salcursor;

end//












drop function if exists calcExp;
delimiter //
create function calcExp(hdate date)
returns int
begin
declare vexp int ;
set vexp = timestampdiff(year,hdate,curdate());
return vexp;

end//
delimiter ;

drop function if exists calcCost;

delimiter //
create function calcCost(nsal double(9,2), hdate date)
returns double(9,2)

begin
declare vexp int;
declare vda double(9,2);
declare vhra double(9,2);
declare vta double(9,2);
declare vspallowance double(9,2);
declare vcost double(9,2);

set vexp = timestampdiff(year,hdate,curdate());

set vda = nsal*0.15;
set vhra = nsal*0.20;
set vta = nsal*0.08;

if vexp >= 1 and vexp<2 then
set vspallowance = nsal*0.10;

elseif vexp>=2 and vexp<4 then
set vspallowance = nsal*0.20;

elseif vexp>=4 then
set vspallowance = nsal*0.30;

else 
set vspallowance = 0;

end if;

set vcost = nsal + vda + vhra + vta + vspallowance;

return vcost;

end//

delimiter ;

drop procedure if exists getstatus;

delimiter //
create procedure getstatus()
begin 
declare vcnt int;

select count(*) into vcnt from emp;

if vcnt = 0 then
signal sqlstate '45000' set message_text = 'Table is empty';

else

select e.empno, e.ename, e.job, e.sal, 
case
when e.sal > (select avg(sal) from emp where deptno = e.deptno) then 'Greater'
when e.sal < (select avg(sal) from emp where deptno = e.deptno) then 'Smaller'
else 'Equal'
end as status, e.deptno from emp e;

end if;

end//

delimiter ;
